title: Reggae One Drop
artist: TabTrainer
about: Leaving space. You rest on beats 1 and 2 and land on beat 3 with the drums. It trains patience and placing a note exactly on time after silence.
level: medium
instrument: bass4
tempo: 76
key: Am
time: 4/4

# Count the rests out loud. The space is part of the groove.
== Learn
level: medium
@Groove
"Am" :h r :q 5.4 :8 7.3 7.2 | :q. 7.2 :8 5.2 :q 7.3 r |
"Am" :h r :q 5.4 :8 7.3 7.2 | :q. 7.2 :8 5.2 :q 7.3 r |
"Dm" :h r :q 5.3 :8 7.2 7.1 | :q. 7.1 :8 5.1 :q 7.2 r |
"G" :h r :q 3.4 :8 5.3 5.2 | :q. 5.2 :8 3.2 :q 5.3 r |
@Ending
"Am" :h r 5.4 |

== Develop
level: medium
# Now the first note comes on the "and" of beat 2, half a beat before the drop.
@Push into beat 3
"Am" :q r :8 r 5.4 :q 5.4 :8 7.3 7.2 | :q. 7.2 :8 5.2 7.3 5.3 :q r |
"Am" :q r :8 r 5.4 :q 5.4 :8 7.3 7.2 | :q. 7.2 :8 5.2 :16 7.3 5.3 8.4 5.4 :q r |
"Dm" :q r :8 r 5.3 :q 5.3 :8 7.2 7.1 | :q. 7.1 :8 5.1 7.2 5.2 :q r |
"G" :q r :8 r 3.4 :q 3.4 :8 5.3 5.2 | :q. 5.2 :8 3.2 5.3 3.3 :q r |
@Ending
"Am" :h r 5.4 |
